/* ===== 计算 ===== */
function compute(r){
  const base=+r.base||0, perf=+r.perf||0, attReq=+r.attReq||24, attAct=+r.attAct||0;
  const comm=+r.comm||0, selfDiff=+r.selfDiff||0, social=+r.social||0, tax=+r.tax||0;
  const gross=(base+perf)/attReq*attAct+comm-selfDiff;
  const net=gross-social-tax;
  return {gross,net,social,tax};
}

const HEAD=['月份','姓名','账号','基本工资','基础绩效','提成工资','自费社保差额','社保扣除','个税扣除','应出勤天数','实际出勤天数'];
function exportCSV(){
  if(!data.length){alert('暂无数据');return}
  const lines=[HEAD.join(',')];
  data.forEach(r=>{
    lines.push([r.month,r.name,r.account,r.base,r.perf,r.comm,r.selfDiff,r.social,r.tax,r.attReq,r.attAct].map(v=>{
      const s=String(v??'');
      return/[",\n]/.test(s)?'"'+s.replace(/"/g,'""')+'"':s;
    }).join(','));
  });
  const blob=new Blob(['\ufeff'+lines.join('\n')],{type:'text/csv;charset=utf-8'});
  const a=document.createElement('a');
  a.href=URL.createObjectURL(blob);
  a.download='工资表_'+new Date().toISOString().slice(0,10)+'.csv';
  a.click();URL.revokeObjectURL(a.href);
}
function importCSV(e){
  const file=e.target.files[0];
  if(!file)return;
  const reader=new FileReader();
  reader.onload=ev=>{
    const text=ev.target.result.replace(/^\ufeff/,'');
    const rows=parseCSV(text);
    if(rows.length<2){alert('文件为空');return}
    let added=0;
    for(let i=1;i<rows.length;i++){
      const c=rows[i];
      if(!c[0]&&!c[1])continue;
      data.push({
        id:'r'+Date.now()+Math.random().toString(36).slice(2,7)+i,
        month:(c[0]||'').trim(),name:(c[1]||'').trim(),account:(c[2]||'').trim(),
        base:+c[3]||0,perf:+c[4]||0,comm:+c[5]||0,selfDiff:+c[6]||0,
        social:+c[7]||0,tax:+c[8]||0,attReq:+c[9]||24,attAct:+c[10]||0
      });
      added++;
    }
    save();renderAll3();
    alert('成功导入 '+added+' 条记录');
    e.target.value='';
  };
  reader.readAsText(file,'UTF-8');
}
function parseCSV(text){
  const rows=[];let row=[],cur='',inQ=false;
  for(let i=0;i<text.length;i++){
    const ch=text[i];
    if(inQ){
      if(ch==='"'){if(text[i+1]==='"'){cur+='"';i++}else inQ=false}
      else cur+=ch;
    }else{
      if(ch==='"')inQ=true;
      else if(ch===','){row.push(cur);cur=''}
      else if(ch==='\n'){row.push(cur);rows.push(row);row=[];cur=''}
      else if(ch!=='\r')cur+=ch;
    }
  }
  if(cur!==''||row.length){row.push(cur);rows.push(row)}
  return rows.filter(r=>r.some(v=>String(v).trim()!==''));
}
